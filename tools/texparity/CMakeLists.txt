cmake_minimum_required(VERSION 3.24)
project(texparity CXX)
# Native texture loader vs the game's d3dx9_38.dll (x86 only: the reference is the 32-bit DLL).
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "texparity must be built for x86")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(BACKEND ${CMAKE_CURRENT_LIST_DIR}/../../backend)
add_executable(texparity texparity.cpp ${BACKEND}/d3dxtex/dxt.cpp ${BACKEND}/d3dxtex/texture.cpp)
target_include_directories(texparity PRIVATE ${BACKEND})
target_link_libraries(texparity PRIVATE d3d9)
target_compile_definitions(texparity PRIVATE _CRT_SECURE_NO_WARNINGS)
target_compile_options(texparity PRIVATE /EHsc /W3 /arch:SSE2)
set_source_files_properties(${BACKEND}/d3dxtex/dxt.cpp ${BACKEND}/d3dxtex/texture.cpp PROPERTIES COMPILE_OPTIONS "/fp:strict")
